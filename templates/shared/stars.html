{# A score out of five, as text rather than as controls. Filled and empty stars are separate spans so the empty ones can be greyed without relying on a second font or an image, and the number is written out for anyone not reading the glyphs. #} {% macro stars(score, label=None) -%} {%- set filled = score|float|round(0, 'common')|int -%} {%- for _ in range(filled) %}★{% endfor -%} {%- if filled < 5 %}{% for _ in range(5 - filled) %}★{% endfor %}{% endif -%} {%- endmacro %} {% macro score_row(label, score) -%}
{{ label }} {{ stars(score, label) }} {{ '%.1f'|format(score|float) }}
{%- endmacro %}
